```python
import csv

import matplotlib
matplotlib.use("Agg")
import torch

from generator.bin_centers import make_bin_centers
from generator.data import load_circles_data, compute_p_real
from generator.mmd import SIGMA_GRID, gaussian_kernel_matrix
from generator.noise import sample_latent
from generator.train import build_generator, train_step
from generator.visualize import ring_band_metrics

EPOCHS = 300  # held fixed at Phase 3's value -- isolates sigma as the one variable
LR = 0.01     # under test (04-CONTEXT.md). Increasing EPOCHS is an explicit escape
BATCH_SIZE = 32  # hatch only if 300 looks undertrained across the board -- not
                 # exercised in this first pass.
RESULTS_DIR = "results"


def main():
    centers = make_bin_centers()
    x_train, _ = load_circles_data()
    p_real = compute_p_real(x_train, centers)

    rows = []
    for sigma in SIGMA_GRID:
        print(f"--- sigma={sigma} ---")
        kernel_matrix = gaussian_kernel_matrix(centers, sigma)
        quantum_layer = build_generator()  # fresh model per sigma -- never reused
        optimizer = torch.optim.Adam(quantum_layer.parameters(), lr=LR)

        for epoch in range(EPOCHS):
            loss = train_step(quantum_layer, optimizer, p_real, kernel_matrix, BATCH_SIZE)
            if epoch % 100 == 0 or epoch == EPOCHS - 1:
                print(f"  epoch {epoch:3d}  loss {loss:.6f}")

        ckpt_path = f"{RESULTS_DIR}/phase4_sigma_{sigma}_checkpoint.pt"
        torch.save(quantum_layer.state_dict(), ckpt_path)

        quantum_layer.eval()
        with torch.no_grad():
            z = sample_latent(1)
            q = quantum_layer(z)[0]
        metrics = ring_band_metrics(q, centers)
        print(f"  ring/gap metrics: {metrics}")
        rows.append({"sigma": sigma, "ring_mass": metrics["ring_mass"], "gap_mass": metrics["gap_mass"]})

    with open(f"{RESULTS_DIR}/phase4_sweep_metrics.csv", "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=["sigma", "ring_mass", "gap_mass"])
        writer.writeheader()
        writer.writerows(rows)


if __name__ == "__main__":
    main()
```
